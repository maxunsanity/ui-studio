<script>
(function () {
  var slides = Array.prototype.slice.call(document.querySelectorAll(".slide"));
  var dots = document.getElementById("dots");
  var count = document.getElementById("count");
  var tocGrid = document.getElementById("tocGrid");
  var toc = document.getElementById("toc");
  var cur = 0;

  slides.forEach(function (s, i) {
    var d = document.createElement("i");
    d.addEventListener("click", function () { go(i); });
    dots.appendChild(d);

    var b = document.createElement("button");
    var t = s.getAttribute("data-t") || ("장 " + (i + 1));
    if (t.indexOf("—") === 0) b.className = "sec";
    b.innerHTML = '<s>' + (i + 1) + '</s><span>' + t + '</span>';
    b.addEventListener("click", function () { toc.classList.remove("on"); go(i); });
    tocGrid.appendChild(b);
  });

  // 끼워 넣은 일정 보드를 칸에 꽉 채운다
  var MIN_W = 1480; // 보드가 짜인 기준 폭
  function fitAll() {
    document.querySelectorAll(".live").forEach(function (live) {
      var body = live.querySelector(".livebody") || live;
      var fit = live.querySelector(".fit");
      var fr = live.querySelector("iframe");
      if (!fit || !fr) return;
      var w = body.clientWidth, h = body.clientHeight;
      if (!w || !h) return;
      // 기준 폭보다 좁으면 줄인다. 위에 탭이 붙어 자리가 16:9가 아니므로 높이도 같이 본다.
      var lw = Math.max(MIN_W, w);
      var lh = Math.round(lw * 9 / 16);
      fr.style.width = lw + "px";
      fr.style.height = lh + "px";
      fit.style.width = lw + "px";
      fit.style.height = lh + "px";
      var sc = Math.min(w / lw, h / lh);
      // 남는 쪽은 가운데로 민다
      fit.style.transform = "translate(" + ((w - lw * sc) / 2) + "px," + ((h - lh * sc) / 2) + "px) scale(" + sc + ")";
    });
  }
  // 보드 자체 제목줄은 덱 안에서 자리만 먹는다. 임베드에서만 감춘다 (파일은 그대로).
  function trimHeader(fr) {
    try {
      // 살아있는 보드 화면에만 먹인다. 인게임 목업은 제 레이아웃이 있어서 건드리면 깨진다.
      if ((fr.getAttribute("data-src") || "").indexOf("board-40") < 0) return;
      var d = fr.contentDocument;
      if (!d || d.getElementById("deckTrim")) return;
      var st = d.createElement("style");
      st.id = "deckTrim";
      st.textContent =
        "header{display:none!important}" +
        "html,body{height:100%}" +
        ".deck{max-width:none!important;height:100%;padding-top:8px!important;display:flex;flex-direction:column}" +
        ".deck-main{flex:1;min-height:0}" +
        ".slide{aspect-ratio:auto!important;height:100%}";
      d.head.appendChild(st);
    } catch (e) { /* 못 건드리면 그냥 둔다 */ }
  }
  window.addEventListener("resize", fitAll);


  // 정적 호스팅이 frame-ancestors로 iframe을 막는다.
  // src로 불러오는 대신 내용을 받아 srcdoc에 심는다 — HTTP 응답이 없어 그 헤더가 안 걸린다.
  function mount(fr) {
    var url = fr.getAttribute("data-src");
    var q = url.split("?")[1] || "";
    var attrs = "";
    if (/bare=1/.test(q)) attrs += ' data-bare="1"';
    var sm = q.match(/side=(fan|ent)/);
    if (sm) attrs += ' data-side="' + sm[1] + '"';
    var vm = q.match(/view=(fan|ent)/);
    if (vm) attrs += ' data-view="' + vm[1] + '"';
    fetch(url.split("?")[0]).then(function (r) { return r.text(); }).then(function (html) {
      if (attrs) html = html.replace("<body>", "<body" + attrs + ">");
      fr.setAttribute("srcdoc", html);
    }).catch(function () {
      fr.setAttribute("src", url);   // 못 받아오면 원래대로 — 크게 열기 버튼은 살아 있다
    });
  }

  function go(i) {
    cur = Math.max(0, Math.min(slides.length - 1, i));
    slides.forEach(function (s, n) { s.classList.toggle("on", n === cur); });
    Array.prototype.forEach.call(dots.children, function (d, n) { d.classList.toggle("on", n === cur); });
    count.textContent = (cur + 1) + " / " + slides.length;
    // 켜진 장에만 일정 보드를 띄운다. 안 보이는 데서 돌릴 이유가 없다.
    var fr = slides[cur].querySelector("iframe[data-src]");
    if (fr && !fr.dataset.mounted) {
      fr.dataset.mounted = "1";
      fr.addEventListener("load", function () { trimHeader(fr); fitAll(); });
      mount(fr);
    }
    fitAll();
    // 인게임 목업은 장이 열릴 때 한 번 저절로 돈다
    var auto = slides[cur].querySelector("iframe[data-autoplay]");
    if (auto) {
      var ping = function () {
        try { auto.contentWindow.postMessage("kpm:play", "*"); } catch (e) {}
      };
      auto.addEventListener("load", function () { setTimeout(ping, 420); }, { once: true });
      if (auto.getAttribute("srcdoc")) setTimeout(ping, 420);
    }
  }

  document.getElementById("prev").addEventListener("click", function () { go(cur - 1); });
  document.getElementById("next").addEventListener("click", function () { go(cur + 1); });
  document.getElementById("tocBtn").addEventListener("click", function () { toc.classList.toggle("on"); });
  toc.addEventListener("click", function (e) { if (e.target === toc) toc.classList.remove("on"); });

  document.addEventListener("keydown", function (e) {
    if (e.key === "ArrowRight" || e.key === "PageDown" || e.key === " ") { go(cur + 1); e.preventDefault(); }
    else if (e.key === "ArrowLeft" || e.key === "PageUp") { go(cur - 1); e.preventDefault(); }
    else if (e.key === "Home") go(0);
    else if (e.key === "End") go(slides.length - 1);
    else if (e.key === "Escape") toc.classList.remove("on");
  });

  go(0);
})();
</script>
