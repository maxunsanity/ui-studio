  function trackOf(cell) { return cell.track ? TRACKS[cell.track] : null; }

  function cellName(cell) {
    if (R57 && cell.r57) return cell.r57.nm;
    if (!cell.track) return cell.name || "";
    if (cell.track === "chance" || cell.track === "practice" || cell.track === "fanfix" || cell.track === "freetime" || cell.track === "camp" || cell.track === "living" || cell.track === "gate" || cell.track === "community" || cell.track === "tax" || cell.track === "gotojail" || cell.track === "util" || cell.track === "corner" || cell.track === "gotojail") {
      return cell.name || "";
    }
    const tr = trackOf(cell);
    const kinds = tr.kinds[tr.ladder[cell.level]];
    return kinds[cell.kind % kinds.length];
  }
  function pickUniqueKind(cell, level) {
    const tr = trackOf(cell);
    const kinds = tr.kinds[tr.ladder[level]];
    const used = {};
    CELLS.forEach(function (c) {
      if (c.track === cell.track && c.id !== cell.id && c.level === level) used[c.kind] = true;
    });
    const free = [];
    for (var i = 0; i < kinds.length; i++) if (!used[i]) free.push(i);
    if (!free.length) return Math.floor(Math.random() * kinds.length);
    return free[Math.floor(Math.random() * free.length)];
  }
  function effectsFor(cell) {
    const tr = trackOf(cell);
    if (!tr) return null;
    const lv = tr.ladder[cell.level];
    const kinds = tr.kinds[lv];
    const now = kinds[cell.kind % kinds.length];
    const midKind = kinds[(cell.kind + 1) % kinds.length];
    const midTxt = tr.mid[cell.kind % tr.mid.length] + " · " + midKind;
    const failTxt = tr.fail[cell.kind % tr.fail.length];
    var daebak;
    if (cell.level >= tr.ladder.length - 1) {
      daebak = "이미 꼭대기 · " + now + " 보너스";
    } else {
      const nextLv = tr.ladder[cell.level + 1];
      const nextKinds = tr.kinds[nextLv];
      // preview unique-ish next
      const used = {};
      CELLS.forEach(function (c) {
        if (c.track === cell.track && c.id !== cell.id && c.level === cell.level + 1) used[c.kind] = true;
      });
      var ni = cell.kind % nextKinds.length;
      for (var i = 0; i < nextKinds.length; i++) {
        if (!used[i]) { ni = i; break; }
      }
      daebak = nextKinds[ni];
    }
    return [
      {k:"대박", c:"daebak", t: daebak},
      {k:"중박", c:"jungbak", t: midTxt},
      {k:"운망", c:"unmang", t: failTxt}
    ];
  }
  function canGrow(cell) {
    const tr = trackOf(cell);
    return !!(tr && cell.level < tr.ladder.length - 1);
  }

  async function grow(id) {
    const cell = CELLS.find(function (c) { return c.id === id; });
    if (!cell || !canGrow(cell) || busy) return;
    busy = true;
    setLandArt("fx-rename.png");
    if (placeTag) placeTag.textContent = "이름 바뀜";
    const tr = trackOf(cell);
    const before = cellName(cell);
    const beforeLv = levelName(cell);
    cell.level += 1;
    cell.kind = pickUniqueKind(cell, cell.level);
    const after = cellName(cell);
    const afterLv = levelName(cell);

    cardBefore.textContent = before + " · " + beforeLv;
    cardAfter.textContent = after;
    cardChip.textContent = beforeLv + " → " + afterLv;
    stamp.classList.remove("show");
    overlay.classList.add("show");
    overlay.setAttribute("aria-hidden", "false");
    await wait(380);
    stamp.classList.add("show");
    await wait(900);
    overlay.classList.remove("show");
    overlay.setAttribute("aria-hidden", "true");
    stamp.classList.remove("show");

    paintCell(cell);
    const el = cellEls[cell.id];
    el.classList.remove("pulse");
    void el.offsetWidth;
    el.classList.add("pulse");
    busy = false;
    show(cell.id);
  }

  function placeOf(cell) {
    if (!cell) return "city";
    if (cell.track === "green") return (cell.level >= 3) ? "arena" : "country";
    if (cell.track === "yellow" && cell.level >= 4) return "arena";
    if (cell.id === 19 || cell.id === 20 || cell.id === 37) return "country";
    return "city";
  }

  function landFileOf(cell) {
    if (!cell) return null;
    if (cell.art === false) return null;   // 시점에서 갈아끼운 칸 — 옛 그림을 쓰지 않는다
    if (cell.track === "yellow") {
      if ((cell.level || 0) === 0) {
        return ["land-y-youtube.png", "land-y-weverse.png", "land-y-short.png"][cell.kind % 3];
      }
      return ["land-y-radio.png", "land-y-cable.png", "land-y-tv.png", "land-y-global.png"][Math.min(cell.level, 4) - 1];
    }
    if (cell.track === "green") {
      if ((cell.level || 0) === 0) {
        return ["land-g-street.png", "land-g-campus.png", "land-g-slot.png"][cell.kind % 3];
      }
      return ["land-g-club.png", "land-g-theater.png", "land-g-hall.png", "land-g-arena.png"][Math.min(cell.level, 4) - 1];
    }
    return LAND_BY_ID[cell.id] || null;
  }

  var landPanTimer = 0;

  function stopLandPan() {
    if (landPanTimer) { clearInterval(landPanTimer); landPanTimer = 0; }

  function startLandPan() {
    stopLandPan();
    var img = document.getElementById("landImg");
    if (!img) return;
    landPanT0 = Date.now();
    img.style.objectPosition = "0% 50%";
    landPanTimer = setInterval(function () {

  function setLandArt(file) {
