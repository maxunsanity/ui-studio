function loop(now){
  if(!running)return;const T=cycleSec()*1000,dispT=Math.min(T,1600);// 화면용: 한 바퀴를 최대 1.6초로 줄여 보여 줌(시안)
  const p=((now-t0)%dispT)/dispT;const bar=document.querySelector("[data-prog]");if(bar)bar.style.width=(p*100).toFixed(1)+"%";
  if(now-(loop.last||0)>dispT){loop.last=now;pop();}
  raf=requestAnimationFrame(loop);
}

function pop(){
  const FACE="⚀⚁⚂⚃⚄⚅",d1=document.querySelector("[data-d1]"),d2=document.querySelector("[data-d2]"),sum=document.querySelector("[data-sum]"),land=document.querySelector("[data-land]");
  const die=document.querySelector("[data-die]");if(die){die.classList.remove("spin");void die.offsetWidth;die.classList.add("spin");}
  const a=1+Math.floor(Math.random()*6),b=1+Math.floor(Math.random()*6); // 주사위 2개 × 6면(board_runtime dice_count 2 · faces 6)
  if(!d1)return;d1.classList.add("roll");d2.classList.add("roll");sum.textContent="…";
  let n=0;const iv=setInterval(()=>{d1.textContent=FACE[Math.floor(Math.random()*6)];d2.textContent=FACE[Math.floor(Math.random()*6)];if(++n>4){clearInterval(iv);
    d1.classList.remove("roll");d2.classList.remove("roll");d1.textContent=FACE[a-1];d2.textContent=FACE[b-1];sum.textContent=a+b;
    pos=((pos-1+a+b)%40)+1;const t=BOARD[pos-1];
    land.innerHTML=`${t[0]}번 · ${esc(t[2])} <em>${esc(t[1])}${t[3]==="corner_jail"?" · 룰(탈출 굴림)":t[4].length?"":" · 보상 없음"}</em>${a===b?' <em style="color:var(--gold)">더블</em>':""}`;
    if(matchMedia("(prefers-reduced-motion: reduce)").matches)return;
    t[4].forEach((k,i)=>{const sl=document.querySelector(`.slot[data-out="${k}"]`);if(sl)setTimeout(()=>fly(sl),120+i*130);});
  }},80);
}
function fly(sl){
  const to=document.querySelector(sl.dataset.to);if(!to)return;
  const a=sl.querySelector(".ic").getBoundingClientRect(),b=to.getBoundingClientRect();
  const el=document.createElement("div");el.className="flyer";el.innerHTML=`<span>${sl.querySelector(".ic").textContent}</span><b>+?×${bet}</b>`;
  const x0=a.left+a.width/2,y0=a.top+a.height/2,x1=b.left+b.width/2,y1=b.top+b.height/2;
  el.style.left=x0+"px";el.style.top=y0+"px";document.body.appendChild(el);
  const k=document.querySelector(".phone").getBoundingClientRect().width/390;el.style.setProperty("--k",k);
  const mx=(x0+x1)/2+(Math.random()*80-40)*k,my=Math.min(y0,y1)-90*k;
  el.animate([{transform:`translate(-50%,-50%) scale(${k})`,offset:0},{transform:`translate(calc(-50% + ${mx-x0}px),calc(-50% + ${my-y0}px)) scale(${1.15*k})`,offset:.45},{transform:`translate(calc(-50% + ${x1-x0}px),calc(-50% + ${y1-y0}px)) scale(${.6*k})`,opacity:.9,offset:1}],{duration:820,easing:"cubic-bezier(.3,.7,.4,1)"}).onfinish=()=>{el.remove();to.classList.remove("bump");void to.offsetWidth;to.classList.add("bump");};
  const q=sl.querySelector(".q");if(q){q.classList.remove("hit");void q.offsetWidth;q.classList.add("hit");}
}

  const b=e.target.closest("button");if(!b)return;
  if(b.dataset.tab){tab=b.dataset.tab;running=false;cancelAnimationFrame(raf);render();}
  else if(b.hasAttribute("data-page")){page=page?0:1;renderTabs();}
  else if(b.hasAttribute("data-run")){running=!running;if(running){t0=performance.now();raf=requestAnimationFrame(loop);}else cancelAnimationFrame(raf);render();if(running)raf=requestAnimationFrame(loop);}
  else if(b.dataset.in){input=b.dataset.in;document.querySelectorAll("[data-in]").forEach(x=>x.classList.toggle("on",x===b));render();}
  else if(b.dataset.bet){bet=Number(b.dataset.bet);render();}
  else if(b.dataset.betmode){betMode=b.dataset.betmode;render();}
