    function pickUniqueKindIdx(levelIdx, exceptId) {
      const lv = LADDER[levelIdx];
      const kinds = KINDS[lv];
      const used = new Set(
        state.slots.filter((x) => x.id !== exceptId && x.levelIdx === levelIdx).map((x) => x.kindIdx % kinds.length)

      void toast.offsetWidth;
      if (msg) toast.classList.add("pop");
    }

    function renderLadder() {
      ladderEl.innerHTML = LADDER.map((n, i) => {
        const used = state.slots.some((s) => s.levelIdx === i);
        return '<span class="' + (used ? "on" : "") + '">' + (i + 1) + ". " + n + "</span>";
      }).join("");
    }

    function showFillers(s) {
      if (!s) {
        rightHint.textContent = "보드에서 칸을 선택하세요.";
        rightEl.innerHTML = '<div class="empty-side">도착한 칸의<br/>대분류 · 소분류 · 지역명</div>';
        return;
      }
      rightHint.textContent =
        "선택 「" + slotLabel(s) + "」 · " + levelName(s.levelIdx) + " · " + state.board;
      const ul = document.createElement("ul");
      ul.className = "filler-list";
      fillersFor(s).forEach((f) => {
        const li = document.createElement("li");
        li.innerHTML =
          '<div class="kind">' + f.kind + "</div>" +
          '<div class="val">' + f.value + "</div>" +
          '<div class="ex">' + (f.ex || "") + "</div>";
        ul.appendChild(li);
      });
      rightEl.innerHTML = "";
      rightEl.appendChild(ul);
    }

    function updateBoardChrome() {
      if (state.board === "Board2") {
        boardBadge.textContent = "Board2 · 국내";
        boardBadge.className = "board-badge b2";
        boardHint.textContent = "국내 풀 · 아레나까지";
      } else {
        boardBadge.textContent = "Board1 · 홍대·수도권";
        boardBadge.className = "board-badge";
        boardHint.textContent = "칸 하나 선택 → 성장";
      }
    }

    function arrive(id) {
      if (state.busy) return;
      state.active = id;
      state.highlightId = null;
      setToast("도착 · 칸 " + id + " 선택");
      render();
      showFillers(state.slots.find((x) => x.id === id));
    }

    function wait(ms) {
      return new Promise((r) => setTimeout(r, ms));
    }

    async function playGrowAnimation(slotId, beforeLabel, afterLabel, afterLevel, doBoard2) {
      const tileEl = tilesEl.querySelector('[data-id="' + slotId + '"]');
      cardBefore.textContent = beforeLabel;
      cardAfter.textContent = afterLabel;
      cardLevel.textContent = "수준 업 · " + afterLevel;
      stamp.classList.remove("show");
      overlay.classList.remove("fly");
      overlay.classList.add("show");
      overlay.setAttribute("aria-hidden", "false");

      await wait(420);
      stamp.classList.add("show");
      await wait(720);

      // Fly into tile
      const cardRect = revealCard.getBoundingClientRect();
      const tileRect = tileEl
        ? tileEl.getBoundingClientRect()
        : { left: window.innerWidth / 2, top: window.innerHeight / 2, width: 40, height: 40 };

      const cx = cardRect.left + cardRect.width / 2;
      const cy = cardRect.top + cardRect.height / 2;
      const tx = tileRect.left + tileRect.width / 2;
      const ty = tileRect.top + tileRect.height / 2;
      const dx = tx - cx;
      const dy = ty - cy;

      overlay.classList.add("fly");
      revealCard.style.transform =
        "translate(" + dx + "px," + dy + "px) scale(0.12)";
      revealCard.style.opacity = "0";

      await wait(560);

      overlay.classList.remove("show", "fly");
      overlay.setAttribute("aria-hidden", "true");
      revealCard.style.transform = "";
      revealCard.style.opacity = "";
      stamp.classList.remove("show");

      state.highlightId = slotId;
      render();
      const grown = tilesEl.querySelector('[data-id="' + slotId + '"]');
      if (grown) {
        grown.classList.add("pulse");
        await wait(200);
      }

      if (doBoard2) {
        boardShell.classList.remove("swap-flash");
        void boardShell.offsetWidth;
        boardShell.classList.add("swap-flash");
        updateBoardChrome();
        setToast("Board2 국내 해금 · 필러 풀 교체 · 소극장→아레나");
        await wait(700);
        boardShell.classList.remove("swap-flash");
      }

      await wait(650);
      state.highlightId = null;
      render();
    }

    async function grow() {
      if (state.busy || !state.active) return;
      const s = state.slots.find((x) => x.id === state.active);
      if (!s) return;

      if (s.levelIdx >= LADDER.length - 1) {
        setToast("이미 최고치 (아레나·대형홀)");
        return;
      }
      if (state.board === "Board1" && s.levelIdx >= 1) {
        setToast("Board1 상한=클럽. 다른 칸을 클럽으로.");
        return;
      }
      if (!canGrow(s)) return;

      state.busy = true;
      btnGrow.disabled = true;

      const beforeLabel = slotLabel(s) + " · " + levelName(s.levelIdx);
      s.levelIdx += 1;
      const lv = levelName(s.levelIdx);
      s.kindIdx = pickUniqueKindIdx(s.levelIdx, s.id);
      const afterLabel = slotLabel(s);
      let unlockedBoard2 = false;
      if (state.board === "Board1" && allClub()) {
        state.board = "Board2";
        unlockedBoard2 = true;
      }

      try {
        await playGrowAnimation(s.id, beforeLabel, afterLabel, lv, unlockedBoard2);
        if (!unlockedBoard2) {
          setToast("칸 " + s.id + " → " + lv + " · 「" + afterLabel + "」");
        }
        showFillers(s);
      } finally {
        state.busy = false;
        render();
        showFillers(s);
      }
    }

    function reset() {
      if (state.busy) return;
      state.board = "Board1";
      state.slots = [
        { id: "A", levelIdx: 0, kindIdx: 0 },
        { id: "B", levelIdx: 0, kindIdx: 1 },
        { id: "C", levelIdx: 0, kindIdx: 2 }
      ];
      state.active = null;
      state.highlightId = null;
      overlay.classList.remove("show", "fly");
      revealCard.style.transform = "";
      revealCard.style.opacity = "";
      setToast("초기화 · 버스킹 3종으로 시작");
      updateBoardChrome();
      render();
      showFillers(null);
    }

    function render() {
      renderLadder();
      updateBoardChrome();
      tilesEl.innerHTML = "";
      const dimOthers = !!state.highlightId;

      state.slots.forEach((s) => {
        const btn = document.createElement("button");
        btn.type = "button";
        btn.className = "tile";
        btn.dataset.id = s.id;
        if (state.active === s.id) btn.classList.add("selected");
        if (dimOthers && state.highlightId !== s.id) btn.classList.add("dim");
        btn.innerHTML =
          '<div class="slot-id">칸 ' + s.id + "</div>" +
          '<div class="venue">' + slotLabel(s) + "</div>" +
          '<span class="chip">' + levelName(s.levelIdx) + "</span>" +
          '<span class="arrive-tag">도착 ✓</span>';
        btn.onclick = () => arrive(s.id);
        tilesEl.appendChild(btn);
      });

      const active = state.slots.find((x) => x.id === state.active);
      btnGrow.disabled = state.busy || !active || !canGrow(active);
    }

    btnGrow.onclick = () => grow();
    btnReset.onclick = () => reset();

    render();
    showFillers(null);
  </script>
</body>
